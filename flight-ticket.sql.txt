/*create table*/
/* Passenger*/ 
CREATE TABLE Passenger (
  passenger_id bigint not null PRIMARY KEY,
  passport_No VARCHAR(10) NOT NULL UNIQUE,
  name VARCHAR(30) NOT NULL,
  gender ENUM('unknown','male','female','non-binary') DEFAULT 'unknown',
  mobile VARCHAR(20)
);  

/*Booking*/
CREATE TABLE Booking (
  booking_No bigint NOT NULL,
  passenger_id bigint,
  booking_price DECIMAL(6,2) NOT NULL check (booking_price > 0),
  booking_time DATETIME NOT NULL, /*booking_time is when a transaction is made, need it to calculate commission fee, the time is usually different from flight time*/
  payment_method ENUM('other','credit card','debit card','paypal','alipay'),
  PRIMARY KEY(booking_No,passenger_id),
  FOREIGN KEY (passenger_id) REFERENCES Passenger (passenger_id)
); 

/*Aircraft*/
CREATE TABLE Aircraft (
  aircraft_type VARCHAR(15) NOT NULL PRIMARY KEY unique
);

/*Seat*/
CREATE TABLE Seat_template_type (
  seat_No VARCHAR(4) NOT NULL,
  aircraft_type VARCHAR(15) not null,
  seat_class ENUM('first-class','economy') NOT NULL,
  PRIMARY KEY (seat_No,aircraft_type),
  FOREIGN KEY (aircraft_type) REFERENCES Aircraft (aircraft_type)/*foreign key constraint*/
); 
/*Airport*/
CREATE TABLE Airport (
  IATA CHAR(3) PRIMARY KEY NOT NULL,
  airport_name VARCHAR(50),
  city VARCHAR(20),
  country_code CHAR(2)
); 
/*Flight*/
CREATE TABLE Flight (
  flight_id bigint NOT NULL unique primary key,/*create to ensure uniqueness*/
  flight_No VARCHAR(10) NOT NULL,
  depart_time DATETIME,
  departAirportIATA CHAR(3),
  arrival_time DATETIME,
  arrivalAirportIATA CHAR(3),
  aircraft_type VARCHAR(15) not null,
  status VARCHAR(10) NOT NULL,
  FOREIGN KEY (aircraft_type) REFERENCES Aircraft (aircraft_type),
  FOREIGN KEY (departAirportIATA) REFERENCES Airport (IATA),
  FOREIGN KEY (arrivalAirportIATA) REFERENCES Airport (IATA),
  constraint check (arrival_time > depart_time)
); 
create table Booking_Seat (
  booking_No bigint,
  flight_id bigint,
  seat_No VARCHAR(4) NULL, /*to show occupied seats, it can be null when passenger has not check-in or assign a seat*/ 
  aircraft_type VARCHAR(15) not null,
  passenger_id bigint not null,
  primary key(booking_No,flight_id,passenger_id),
  FOREIGN KEY (aircraft_type) REFERENCES Aircraft (aircraft_type),
  foreign key(booking_No,passenger_id) references Booking(booking_No,passenger_id),
  foreign key(flight_id) references Flight(flight_id) on delete cascade,
  foreign key(seat_No,aircraft_type) references Seat_template_type(seat_No,aircraft_type),/*foreign key constraint：seat_No assign must exist in Seat_template_type with the same aircraft_type*/
  unique (flight_id, seat_no)
);
commit;

CREATE TRIGGER prevent_occupied_seat_update
BEFORE UPDATE ON Booking_Seat
FOR EACH ROW
BEGIN
    IF NEW.seat_No IS NOT NULL AND (NEW.seat_No <> OLD.seat_No OR OLD.seat_No IS NULL) THEN/*check only when successfully update a seat + seat_No change from either old.seat or null*/ 
        IF EXISTS (
            SELECT 1
            FROM Booking_Seat BS 
            where BS.seat_No = NEW.seat_No 
              AND BS.flight_id=NEW.flight_id
              AND BS.booking_No <> NEW.booking_No  /*excluded current update*/
        ) THEN
            SIGNAL SQLSTATE 'NNNNN'
            SET MESSAGE_TEXT = 'warning!！！seat is already occupied!!!';
        END IF;
    END IF;
END;

/*a)	passenger user: flight search: From Chongqing to Sanya between 2026-02-16 to 2026-02-26*/
select 
  F.depart_time, 
  F.arrival_time, 
  F.flight_No, 
  total.totalticket, 
  COALESCE(sub.ticketsold, 0) as ticketsold,/*if sub.ticketsold is null, replace with 0*/
  (total.totalticket-COALESCE(sub.ticketsold, 0)) as available
from Flight F
join Airport A1 ON F.DepartAirportIATA=A1.IATA
join Airport A2 ON F.ArrivalAirportIATA=A2.IATA
join (
  select aircraft_type, count(*) AS totalticket/*totalticket=all available seats count for flight's aircraft_type*/
  from Seat_template_type S
  group by aircraft_type
) as total ON F.aircraft_type=total.aircraft_type
left join (
  select flight_id, count(*) AS ticketsold/*ticketsold=flight_id counts in Booking_Seat Table*/
  from Booking_Seat BS
  group by flight_id
) as sub ON sub.flight_id=F.flight_id
where A1.city='Chongqing'
  and A2.city='Sanya'
  and F.depart_time>='2026-02-16 00:00:00' 
  and F.depart_time<'2026-02-27 00:00:00'
  and (total.totalticket-COALESCE(sub.ticketsold, 0)) >0; /*ignore full flights*/

/*c)	airline user: Sichuan Airline checks passengers who not yet assign a seat or check-in, and show the remaining first-class seats for these passenger, example:flight_id101*/
WITH FlightAircraft AS (
    SELECT aircraft_type 
    FROM Flight 
    WHERE flight_id = '101'
),
SeatSummary AS (
    SELECT 
        COUNT(S.seat_No) AS totalfirstclass,
        COUNT(BS.seat_No) AS occupiedfirstclass,
        (COUNT(S.seat_No) - COUNT(BS.seat_No)) AS remainingfirstclass
    FROM Seat_template_type S
    LEFT JOIN Booking_Seat BS 
        ON BS.seat_No = S.seat_No
        AND BS.aircraft_type = (SELECT aircraft_type FROM FlightAircraft)
        AND BS.flight_id = '101'
    WHERE S.aircraft_type = (SELECT aircraft_type FROM FlightAircraft)
        AND S.seat_class = 'first-class'
    HAVING remainingfirstclass > 0
),
UnassignedPassengers AS (
    SELECT P.name
    FROM Booking_Seat BS
    JOIN Booking B ON B.booking_No = BS.booking_No and B.passenger_id=BS.passenger_id
    JOIN Passenger P ON P.passenger_id = B.passenger_id
    WHERE BS.seat_No IS NULL 
        AND BS.flight_id = '101'
),
AvailableSeats AS (
    SELECT S.seat_No AS availablefirstclassseat
    FROM Seat_template_type S
    WHERE S.aircraft_type = (SELECT aircraft_type FROM FlightAircraft)
        AND S.seat_class = 'first-class'
        AND NOT EXISTS (
            SELECT 1 
            FROM Booking_Seat BS 
            WHERE BS.seat_No = S.seat_No 
                AND BS.flight_id = '101'
        )
)

SELECT 
    (SELECT remainingfirstclass FROM SeatSummary) AS remaining_fseats,
    (SELECT GROUP_CONCAT(name) FROM UnassignedPassengers) AS unassigned_passengers,
    (SELECT GROUP_CONCAT(availablefirstclassseat) FROM AvailableSeats) AS available_fseats_list;

/*d)	intermediary user: calculate total 2026 Feb commission fee, and list every commission fee order by transaction time and price. */
select SUM(booking_price)*0.1 AS Commission202602_pound /*calculate commission fee on-demand to reduce system burden*/
from Booking
where booking_time>='2025-02-01' and booking_time<='2025-02-28'
group by month(booking_time) /*sum table*/
having SUM(booking_price)*0.1>0; /*when create table already check booking_price>0, double check commission fee here*/
select date(booking_time), booking_price
from Booking
where booking_time>='2025-02-01' and booking_time<='2025-02-28'
order by booking_time asc,booking_price desc; /*order for clarity, list all sum transaction*/

/*e)	a useful survey: count payment method per transaction, list most popular one.*/
with payment_count as (
  select 
    sum(case when payment_method='credit card' then 1 else 0 end) as creditcard, /*when case, mark 1, else 0, sum all marks, end. instead of using count.'count' clause counts all not null values.*/
    sum(case when payment_method='debit card' then 1 else 0 end) as debitcard,
    sum(case when payment_method='paypal' then 1 else 0 end) as paypal,
    sum(case when payment_method='Alipay' then 1 else 0 end) as Alipay,
    sum(case when payment_method='other' then 1 else 0 end) as other
  from Booking
)
select
  case greatest(creditcard,debitcard,paypal,Alipay,other)
    when creditcard then 'credit card'
    when debitcard then 'debit card'
    when paypal then 'paypal'
    when Alipay then 'Alipay'
    else 'other'
  end as most_popular,/*only output the greatest method*/
  greatest(creditcard,debitcard,paypal,Alipay,other) as most_popular_count /*only output the greatest count.*/
from payment_count; 
  
/*f)	flight ticket promotion: inactive passengers, targeting no booking passengers, retrieve mobile to send coupon*/
